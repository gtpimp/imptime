/etc/nginx/sites-available/imptime_maintenance.conf
--------------
server {

     listen 443;
     server_name maintenance.*;

     root /opt/imptime_maintenance/www;

     client_max_body_size 4G;

     access_log /var/log/nginx/imptime-maintenance-access.443.log;
     error_log /var/log/nginx/imptime-maintenance-error.443.log;

     location / {
        #try_files $uri /index.html;
        try_files $uri $uri/ /index.html =404;
     }

     ssl on;
     ssl_certificate /etc/letsencrypt/live/maintenance.careerimptime.com/fullchain.pem;
     ssl_certificate_key /etc/letsencrypt/live/maintenance.careerimptime.com/privkey.pem;
     ssl_session_timeout 5m;
     ssl_protocols TLSv1.2;
     ssl_ciphers "EECDH+AESGCM:EDH+AESGCM:AES256+EECDH:AES256+EDH";
     ssl_prefer_server_ciphers on;
}
